`timescale 1ns/1ps

module fulladder64_test;

reg [63:0] a;
reg [63:0] b;
reg cin;

wire [63:0] sum;
wire cout;

fulladder64 uut (
    .a(a),
    .b(b),
    .cin(cin),
    .sum(sum),
    .cout(cout)
);

initial begin

    $monitor("Time = %0t | a = %b | b = %b | cin = %b | sum = %b | cout = %b",
             $time, a, b, cin, sum, cout);

    // Test 1
    a   = 64'h0000_0000_FFFF_FFFF;
    b   = 64'h0000_0000_0000_0001;
    cin = 0;
    #10;

    // Test 2
    a   = 64'hFFFF_FFFF_FFFF_FFFF;
    b   = 64'h0000_0000_0000_0000;
    cin = 1;
    #10;

    // Test 3
    a   = 64'h7FFF_FFFF_FFFF_FFFF;
    b   = 64'h0000_0000_0000_0001;
    cin = 0;
    #10;

    // Test 4
    a   = 64'h8000_0000_0000_0000;
    b   = 64'h8000_0000_0000_0000;
    cin = 0;
    #10;

    // Test 5
    a   = 64'hFFFF_FFFF_FFFF_FFFF;
    b   = 64'hFFFF_FFFF_FFFF_FFFF;
    cin = 1;
    #10;

    // Test 6
    a   = 64'h0000_0000_0000_0000;
    b   = 64'h0000_0000_0000_0000;
    cin = 0;
    #10;

    $finish;

end

endmodule
